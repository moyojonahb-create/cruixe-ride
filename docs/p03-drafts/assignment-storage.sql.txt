 id uuid PRIMARY KEY,
 ride_id uuid NOT NULL UNIQUE REFERENCES public.rides(id),
 driver_id uuid NOT NULL REFERENCES public.drivers(id),
 offer_id uuid NOT NULL REFERENCES public.offers(id),
 amount_minor bigint NOT NULL CHECK (amount_minor > 0),
 currency text NOT NULL CHECK (currency ~ '^[A-Z]{3}$'),
 -- This format check does NOT authorize a currency/exponent; registry gate needed.

 UNIQUE(id,ride_id,driver_id)

-- Draft 02: active row only, no Redis/presence TTL expiration.
CREATE TABLE ride_assignment_private.active_driver_reservations (

 FOREIGN KEY(assignment_id,ride_id,driver_id)
  REFERENCES ride_assignment_private.accepted_agreements(id,ride_id,driver_id)
);
-- Draft 03: shared actor-scoped command receipt; coordinate with P0.5 before DDL.
CREATE TABLE ride_assignment_private.command_receipts (

 request_hash bytea NOT NULL CHECK(octet_length(request_hash)=32),
 assignment_id uuid REFERENCES ride_assignment_private.accepted_agreements(id),
 result jsonb,
 state text NOT NULL CHECK(state IN ('pending','committed')),
 created_at timestamptz NOT NULL,
 PRIMARY KEY(actor_id,operation,idempotency_key),
 CHECK(state <> 'committed' OR result IS NOT NULL)
);
